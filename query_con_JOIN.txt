1. Selezionare tutti gli studenti iscritti al Corso di Laurea in Economia
SELECT * FROM students s JOIN degrees d ON s.degree_id = d.id WHERE d.id = 53;

2. Selezionare tutti i Corsi di Laurea Magistrale del Dipartimento di
Neuroscienze
SELECT * FROM departments dep JOIN degrees deg ON dep.id = deg.department_id WHERE deg.department_id = 7 AND deg.level = 'magistrale';

3. Selezionare tutti i corsi in cui insegna Fulvio Amato (id=44)
SELECT * FROM courses a JOIN course_teacher B ON a.id = b.course_id WHERE b.teacher_id = 44;

4. Selezionare tutti gli studenti con i dati relativi al corso di laurea a cui sono iscritti e il relativo dipartimento, in ordine alfabetico per cognome e nome
SELECT 
    CONCAT(a.name, ' ', a.surname) AS `Studente`,
    c.name AS `Dipartimento`,
    b.name AS `Corso di Laurea`,
    b.level AS `Ciclo`,
    b.website AS `Sito web`
FROM
    students a
        JOIN
    degrees b ON a.degree_id = b.id
        JOIN
    departments c ON b.department_id = c.id
ORDER BY a.surname;

5. Selezionare tutti i corsi di laurea con i relativi corsi e insegnanti
SELECT 
    a.name AS `CdL`,
    b.name AS `Corso`,
    CONCAT(d.name, ' ', d.surname) AS `Insegnante`
FROM
    degrees a
        JOIN
    courses b ON a.id = b.degree_id
        JOIN
    course_teacher c ON b.id = c.course_id
        JOIN
    teachers d ON c.teacher_id = d.id
ORDER BY a.name ASC;


6. Selezionare tutti i docenti che insegnano nel Dipartimento di Matematica (54)
SELECT 
    d.*
FROM
    degrees a
        JOIN
    courses b ON a.id = b.degree_id
        JOIN
    course_teacher c ON b.id = c.course_id
        JOIN
    teachers d ON c.teacher_id = d.id
    WHERE a.department_id = 5
    GROUP BY d.id;
